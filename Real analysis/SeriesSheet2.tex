\section*{Convergence of series}
\begin{enumerate}
    \item Let \( \sum_{n=m}^{\infty} a_n \) be a formal series of real numbers. Then \( \sum_{n=m}^{\infty} a_n \) converges if and only if, for every real number \( \varepsilon > 0 \), there exists an integer \( N \geq m \) such that
    \[
    \left| \sum_{n=p}^{q} a_n \right| \leq \varepsilon \quad \text{for all } p,q \geq N.
    \]
    $\textbf{Solution.}$
    \begin{enumerate}
        \item[$(\implies)$] If $\sum_{n=m}^{\infty} a_n$ converges then, by definition, the partial sum sequence 
        \[
        (S_N)_{N=m}^{\infty} = \left(\sum_{n=m}^{N}a_n\right)_{N=m}^{\infty}
        \]
        converges to $L$.
        Therefore, for any $\varepsilon > 0$ there exists some $M \geq m$ such that for any $p-1,q \geq M$
        \[
        \vert S_q - S_{p-1} \vert \leq \varepsilon.
        \]
        Without loss of generality assume $p - 1 \leq q$, it follows that
        \[
        \left\lvert \sum_{n=m}^{q}a_n - \sum_{n=m}^{p-1}a_n \right\rvert = \left\lvert \sum_{n=p}^{q}a_n \right\rvert \leq \varepsilon.
        \]
        
        \item[$(\impliedby)$] Doing the reverse process of the direct implication the result follows.
    \end{enumerate}
    \begin{flushright}
        \qed
    \end{flushright}

    \item \textbf{(Zero test)}. Let \( \sum_{n=m}^{\infty} a_n \) be a convergent series of real numbers. Then we must have \( \lim_{n \to \infty} a_n = 0 \). To put this another way, if \( \lim_{n \to \infty} a_n \) is non-zero or divergent, then the series \( \sum_{n=m}^{\infty} a_n \) is divergent.
    \\$\textbf{Solution.}$
    If $\sum_{n=m}^{\infty}a_n$ converges then by last problem, for any $\varepsilon > 0$, there exists some $M \geq m$ such that for any $p,q \geq M$
    \begin{equation}
        \left \lvert \sum_{n=p}^{q} a_n \right\rvert \leq \varepsilon.
    \end{equation}
    Suppose for the sake of contradiction that $\lim_{n\to\infty}a_n = L > 0$ so that for any $\varepsilon > 0$ there exists 
    some $M'$ such that for any $n \geq M'$ 
    \[
    \vert a_n - L \vert < \varepsilon
    \]
    so that in particular $a_n \geq L-\varepsilon$.

    Therefore, for any $p,q \geq K := \max\{M, M'\}$
    \[
    \left \lvert \sum_{n=p}^{q} a_n \right\rvert \geq \left| \sum_{n=p}^{q} L- \varepsilon \right| = (q-p+1)\left| L - \varepsilon \right|.
    \]
    Hence, let $\varepsilon = L/2$, we see that if the above inquality were true 
    \[
    \left \lvert \sum_{n=p}^{q} a_n \right\rvert \geq \frac{(q-p+1)L}{2}.
    \]
    Then let $q = p+1$, it follows that 
    \[
        \left \lvert \sum_{n=p}^{q} a_n \right\rvert \geq L > \varepsilon. 
    \]
    A contradiction of inequality $(1)$.
    \begin{flushright}
        \qed
    \end{flushright}

    \item \textbf{(Absolute convergence test)}. Let \( \sum_{n=m}^{\infty} a_n \) be a formal series of real numbers. If this series is absolutely convergent, then it is also conditionally convergent. Furthermore, in this case we have the triangle inequality
    \[
    \left| \sum_{n=m}^{\infty} a_n \right| \leq \sum_{n=m}^{\infty} |a_n|.
    \]
    $\textbf{Solution.}$
    If $\sum_{n=m}^{\infty}\vert a_n \vert$ converges to $L$, then for any $\varepsilon > 0$ there exists some 
    $M \geq m$ such that for any $p,q \geq M$
    \[
    \left\lvert \sum_{n=p}^{q}\vert a_n \vert \right\rvert \leq \varepsilon.
    \]
    Hence, 
    \[
    \sum_{n=p}^{q}\vert a_n \vert \leq \varepsilon.
    \]
    By triangle inequality over finite series, for any $p,q \geq M$ we have
    \[
    \left\lvert \sum_{n=p}^{q} a_n \right\rvert \leq \sum_{n=p}^{q}\vert a_n \vert.
    \]
    We conclude that $\sum_{n=p}^{q}a_n$ converges. Trivially, we also note that again, by triangle inequality over 
    finite series, each partial sum $S_N = \sum_{n=m}^{N}a_n$ is smaller than or equal to $T_N = \sum_{n=m}^{N}\vert a_n \vert$,
    meaning that by comparison principle $\lim_{N\to\infty}S_N \leq \lim_{N\to\infty}T_N$.
    \begin{flushright}
        \qed
    \end{flushright}

    \item (\textbf{Series laws}).
    \begin{enumerate}
        \item[(a)] If \( \sum_{n=m}^{\infty} a_n \) is a series of real numbers converging to \( x \), and
        \( \sum_{n=m}^{\infty} b_n \) is a series of real numbers converging to \( y \), then
        \( \sum_{n=m}^{\infty} (a_n + b_n) \) is also a convergent series, and converges to \( x + y \).\\
        In particular, we have
        \[
        \sum_{n=m}^{\infty} (a_n + b_n)
        = \sum_{n=m}^{\infty} a_n + \sum_{n=m}^{\infty} b_n.
        \]

        \item[(b)] If \( \sum_{n=m}^{\infty} a_n \) is a series of real numbers converging to \( x \), and
        \( c \) is a real number, then \( \sum_{n=m}^{\infty} (c a_n) \) is also a convergent series, and
        converges to \( cx \).\\
        In particular, we have
        \[
        \sum_{n=m}^{\infty} (c a_n)
        = c \sum_{n=m}^{\infty} a_n.
        \]

        \item[(c)] Let \( \sum_{n=m}^{\infty} a_n \) be a series of real numbers, and let
        \( k \geq 0 \) be an integer. If one of the two series
        \( \sum_{n=m}^{\infty} a_n \) and \( \sum_{n=m+k}^{\infty} a_n \) are convergent,
        then the other one is also, and we have the identity
        \[
            \sum_{n=m}^{\infty} a_n
            = \sum_{n=m}^{m+k-1} a_n + \sum_{n=m+k}^{\infty} a_n.
        \]

        \item[(d)] Let \( \sum_{n=m}^{\infty} a_n \) be a series of real numbers converging to \( x \),
        and let \( k \) be an integer. Then
        \( \sum_{n=m+k}^{\infty} a_{n-k} \) also converges to \( x \).
    \end{enumerate}
    $\textbf{Solution.}$
    Let $T_N = \sum_{n=m}^{N}a_n$ and $R_N = \sum_{n=m}^{N}b_n$.
    \begin{enumerate}
        \item
        The partial sum of the series $\sum_{n=m}^{\infty}a_n+b_n$
        is trivially
        \[
        \sum_{n=m}^{N} (a_n + b_n) =  T_N + R_N.
        \]
        we also note that by limit laws 
        \[
        \lim_{n\to\infty}T_N+R_N = \lim_{n\to\infty}T_N + \lim_{n\to\infty}R_N.
        \]
        We conclude that 
        \[
        \sum_{n=m}^{\infty}(a_n+b_n) = \lim_{N\to\infty} S_N = \lim_{N\to\infty} T_N + R_N = x+y.
        \] 

        \item Similarly to the item before, due to the linear properties of both limit and partial sums we get 
        \[
        \sum_{n=m}^{N} ca_n = cT_N
        \]
        so that $\lim_{N\to\infty} S_N = \lim_{N\to\infty}cT_N = c\lim_{N\to\infty}T_N = cx$.

        \item If $\sum_{n=m}^{\infty}a_n$ converges then
        the first claim is trivial, follows from convergent limit properties.
        

        

    \end{enumerate}

    \item \textbf{(Telescoping series)}. Let \( (a_n)_{n=0}^{\infty} \) be a sequence of real numbers which converge to 0, i.e., \( \lim_{n \to \infty} a_n = 0 \). Then the series
    \[
    \sum_{n=0}^{\infty} (a_n - a_{n+1})
    \]
    converges to \( a_0 \).\\
    $\textbf{Solution.}$
    We see that for any partial sum $S_N$ starting from $0$, 
    \[
    \sum_{n=0}^{N}a_n - a_{n+1} = \sum_{n=0}^{N}a_n - \sum_{n=0}^{N}a_{n+1}
    \]
    doing a substitution $n \rightarrow n + 1$ we get
    \[
    S_N = \sum_{n=0}^{N}a_n - \sum_{n=1}^{N+1}a_{n} = a_0 + \sum_{n=1}^{N}a_n - \sum_{n=1}^{N}a_n + a_{N+1}
    \]
    therefore, 
    \[
    S_N = a_0 - a_{N+1}.
    \]
    We notice since $(a_n)$ converges and by using limit laws that
    \[
    \lim_{N\to\infty}S_N = \lim_{N\to\infty} a_0 - \lim_{N\to\infty}a_{N+1} = a_0.
    \]
    \begin{flushright}
        \qed
    \end{flushright}

    \item (\textbf{Alternating series test.}) Let $(a_n)_{n=m}^{\infty}$ be a sequence of real numbers which are non-negative and decreasing, 
    thus $a_n \geq 0$ and $a_n \geq a_{n+1}$ for every $n \geq m$. 
    Then the series $\sum_{n=m}^{\infty}(-1)^n a_n$ is convergent if and only if the 
    sequence $a_n$ converges to $0$ as $n \to \infty$.\\
    $\textbf{Proof.}$
    \begin{enumerate}
        \item[$(\implies)$] If \( \sum_{n=m}^{\infty} (-1)^n a_n \) is convergent, then the sequence $((-1)^n a_n)_{n=m}^{\infty}$ converges to zero.
        By zero test for sequences, we know that the sequence $(\vert (-1)^n a_n \vert)_{n=m}^{\infty}$ converges to zero, and thus $(a_n)_{n=m}^{\infty}$ converges to zero.

        \item[$(\impliedby)$] 
        Let any partial sum be $S_N = \sum_{n=m}^{N}(-1)^n a_n$.
        Let $N$ be an even number. We claim that for any $k \geq 0$ 
        \[
        S_{N+2k} \leq S_N.
        \]
        To prove this claim we induct on $k$.
        \begin{enumerate}
            \item[\textbf{Basis.}]
            If $k = 0$ then 
            \[
                S_{N + 2(0)} = S_N \leq S_N. 
            \]

            \item[\textbf{Hypothesis.}]
            Assume that for any positive integer $k$ it follows that 
            \[
                S_{N+2k} \leq S_{N}.
            \]

            \item[\textbf{Thesis.}]
            Our goal is to prove that for the integer $k+1$ it follows that 
            \[
                S_{N+2(k+1)} \leq S_N.
            \]
            The following chain of equalities will hold, by the recursive definition of a partial sum
            \[
                S_{N+2k+2} = S_{N + 2k} + (-1)^{N+2k+1}(a_{N+2k+1}) + (-1)^{N+2k+2}(a_{N+2k+2}).
            \]
            Thus since $N$ is even
            \[
            S_{N+2k+2} = S_{N+2k} + (-1)(a_{N+2k+1}) + a_{N+2k+2}
            \]
            note that $a_{N+2k+2} - a_{N+2k+1} \leq 0$ and thus
            \[
            S_{N+2k+2} \leq S_{N+2k}.
            \]
            By hypothesis and by transitivity we conclude that $S_{N+2k+2} \leq S_N$.
        \end{enumerate}
        Seeing $S_{N+2k}$ as a sequence when varying $k$, the previous result allows us to use it as a lower bound for $S_N$ for any $N \geq m$.
        Consider now the dual odd sequence $S_{N+2k+1}$, for any $k \geq 0$
        we observe that 
        \[
        S_{N+2k+1} = S_{N+2k} - a_{N+2k+1}
        \]
        thus $S_{N+2k+1} \leq S_{N+2k}$.
        Furthermore we claim that 
        \[
        S_{N+2k+1} \geq S_{N+1}.
        \]
        To prove this claim, by induction over $k$
        \begin{enumerate}
            \item[\textbf{Basis.}]
            If $k = 0$ then 
            \[
            S_{N+1} \geq S_{N+1}
            \]
            so this case holds.
            \item[\textbf{Hypothesis.}]
            Assume that for any $k \geq 0$ it holds 
            \[
            S_{N+2k+1} \geq S_{N+1}.
            \]
            \item[\textbf{Thesis.}]
            Our goal is to prove that for the positive integer $k+1$
            \[
            S_{N+2(k+1) + 1 } \geq S_{N+1}
            \]
            holds. 
            To verify this
            \[
            S_{N+2k+3} = S_{N+2k+1} + (a_{N+2k+2} - a_{N+2k+3}) \geq S_{N+2k+1}. 
            \]
            By inductive hypothesis and by transitivity we get
            \[
            S_{N+2k+3} \geq S_{N+2k+1} \geq S_{N+1}.
            \] 
        \end{enumerate}
        We can arrange our sequences into 
        \[
        S_{N+2k} \geq S_{N+2k+1} \geq S_{N+1}
        \]
        and therefore 
        \[
        S_{N} + a_{N+1} \geq S_{N+2k+1} \geq S_{N}-a_{N+1}.
        \]
        So 
        \[
        \vert S_{N+2k+1} - S_{N} \vert \leq a_{N+1}
        \]
        which implies that 
        \[
        \left \lvert \sum_{n=N+1}^{2k+1} (-1)^n a_n \right \rvert \leq a_{N+1}.
        \]
        Since $a_{N+1}$ converges to zero and is decreasing, for any $\varepsilon > 0$ we can find some $M \geq m$ such that for any $n \geq M$
        $a_{n} \leq \varepsilon$.
        By problem (1), since $N+1$ and $2k+1$ are arbitrary, this series converges.
        \begin{flushright}
            \qed
        \end{flushright}

    \item \textbf{(Comparison test.)}
    Let $\sum_{n=m}^{\infty} a_n$ and $\sum_{n=m}^{\infty} b_n$ be two formal series of real numbers, and suppose that $|a_n| \leq b_n$ for all $n \geq m$. Then if $\sum_{n=m}^{\infty} b_n$ is convergent, then $\sum_{n=m}^{\infty} a_n$ is absolutely convergent, and in fact
    \[
    \left| \sum_{n=m}^{\infty} a_n \right| \leq \sum_{n=m}^{\infty} |a_n| \leq \sum_{n=m}^{\infty} b_n.
    \]
    $\textbf{Proof.}$
    Since $(b_n)$ is convergent, the partial sum sequence
    \[
    (S_N)_{N=m}^{\infty} = \left(\sum_{n=m}^{N} b_n\right)_{N=m}^{\infty} 
    \]
    is bounded by a positive real number $K$ since it converges and
    $0 \leq \vert a_n \vert \leq b_n$.

    The partial sum sequence \( (T_N)_{N=m}^{\infty} =  \left(\sum_{n=m}^{N} \vert a_n\vert \right)_{N=m}^{\infty} \) is a positive terms sum and therefore it is increasing.
    Notice that each of the partial sums will be less than or equal to the partial sums $S_N$ (over the corresponding indices).
    
    Since it is increasing and it is bounded above by $K$ it converges, and it will converge to a smaller limit to that of $S_N$.
    \begin{flushright}
        \qed
    \end{flushright}
